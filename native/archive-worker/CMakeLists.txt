cmake_minimum_required(VERSION 3.20)
project(vesperwind_archive C)
set(CMAKE_C_STANDARD 11)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MSVC_USE_STATIC_CRT ON CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
# libarchive's configure macros re-evaluate include paths. Use forward slashes
# so Windows paths such as C:\Users never become CMake escape sequences.
file(TO_CMAKE_PATH "${LIBARCHIVE_SOURCE}" LIBARCHIVE_SOURCE)
file(TO_CMAKE_PATH "${ZLIB_SOURCE}" ZLIB_SOURCE)
add_subdirectory("${ZLIB_SOURCE}" zlib EXCLUDE_FROM_ALL)
set(ZLIB_LIBRARY zlibstatic CACHE STRING "" FORCE)
set(ZLIB_INCLUDE_DIR "${ZLIB_SOURCE};${CMAKE_CURRENT_BINARY_DIR}/zlib" CACHE STRING "" FORCE)
set(ENABLE_ZLIB ON CACHE BOOL "" FORCE)
# Never enter libarchive's optional PCRE/MinGW libgcc probes on MSVC.
set(POSIX_REGEX_LIB LIBC CACHE STRING "" FORCE)
foreach(option BZip2 LZMA ZSTD LZ4 LZO LIBB2 OPENSSL MBEDTLS NETTLE LIBXML2 EXPAT PCREPOSIX PCRE2POSIX ICONV WIN32_XMLLITE LIBGCC CNG)
  set(ENABLE_${option} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(option TEST INSTALL TAR CPIO CAT UNZIP)
  set(ENABLE_${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${LIBARCHIVE_SOURCE}" libarchive EXCLUDE_FROM_ALL)
add_executable(vesperwind-archive main.c)
target_compile_definitions(vesperwind-archive PRIVATE LIBARCHIVE_STATIC)
target_link_libraries(vesperwind-archive PRIVATE archive_static zlibstatic)
add_executable(archive-space-test space_test.c)
target_compile_definitions(archive-space-test PRIVATE LIBARCHIVE_STATIC)
target_link_libraries(archive-space-test PRIVATE archive_static zlibstatic)
if(MSVC)
  set_property(TARGET vesperwind-archive archive-space-test archive_static zlibstatic PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
