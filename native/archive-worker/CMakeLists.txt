cmake_minimum_required(VERSION 3.20)
project(vesperwind_archive C)
set(CMAKE_C_STANDARD 11)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(MSVC_USE_STATIC_CRT ON CACHE BOOL "" FORCE)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
# libarchive's configure macros re-evaluate include paths. Use forward slashes
# so Windows paths such as C:\Users never become CMake escape sequences.
file(TO_CMAKE_PATH "${LIBARCHIVE_SOURCE}" LIBARCHIVE_SOURCE)
file(TO_CMAKE_PATH "${ZLIB_SOURCE}" ZLIB_SOURCE)
file(TO_CMAKE_PATH "${XZ_SOURCE}" XZ_SOURCE)
foreach(option NLS DOC DOXYGEN EXTERNAL_SHA256 MICROLZMA_ENCODER MICROLZMA_DECODER LZIP_DECODER TOOL_XZ TOOL_XZDEC TOOL_LZMADEC TOOL_LZMAINFO TOOL_SCRIPTS)
  set(XZ_${option} OFF CACHE BOOL "" FORCE)
endforeach()
set(XZ_THREADS no CACHE STRING "" FORCE)
set(XZ_ENCODERS "lzma1;lzma2" CACHE STRING "" FORCE)
set(XZ_DECODERS "lzma1;lzma2;x86;delta" CACHE STRING "" FORCE)
set(XZ_CHECKS "crc32" CACHE STRING "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
add_subdirectory("${XZ_SOURCE}" xz EXCLUDE_FROM_ALL)
target_include_directories(liblzma INTERFACE "${XZ_SOURCE}/src/liblzma/api")
set(ENABLE_LZMA ON CACHE BOOL "" FORCE)
add_subdirectory("${ZLIB_SOURCE}" zlib EXCLUDE_FROM_ALL)
set(ZLIB_LIBRARY zlibstatic CACHE STRING "" FORCE)
set(ZLIB_INCLUDE_DIR "${ZLIB_SOURCE};${CMAKE_CURRENT_BINARY_DIR}/zlib" CACHE STRING "" FORCE)
set(ENABLE_ZLIB ON CACHE BOOL "" FORCE)
# Never enter libarchive's optional PCRE/MinGW libgcc probes on MSVC.
set(POSIX_REGEX_LIB LIBC CACHE STRING "" FORCE)
foreach(option BZip2 ZSTD LZ4 LZO LIBB2 OPENSSL MBEDTLS NETTLE LIBXML2 EXPAT PCREPOSIX PCRE2POSIX ICONV WIN32_XMLLITE LIBGCC CNG)
  set(ENABLE_${option} OFF CACHE BOOL "" FORCE)
endforeach()
foreach(option TEST INSTALL TAR CPIO CAT UNZIP)
  set(ENABLE_${option} OFF CACHE BOOL "" FORCE)
endforeach()
add_subdirectory("${LIBARCHIVE_SOURCE}" libarchive EXCLUDE_FROM_ALL)
target_include_directories(archive_static PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
if(WIN32)
  # libarchive uses the OS RNG even with its optional CNG crypto formats disabled.
  target_link_libraries(archive_static bcrypt)
endif()
add_executable(vesperwind-archive main.c archive_memory.c)
target_compile_definitions(vesperwind-archive PRIVATE LIBARCHIVE_STATIC)
target_link_libraries(vesperwind-archive PRIVATE archive_static zlibstatic liblzma)
add_executable(archive-space-test space_test.c archive_memory.c)
target_compile_definitions(archive-space-test PRIVATE LIBARCHIVE_STATIC)
target_link_libraries(archive-space-test PRIVATE archive_static zlibstatic liblzma)
add_executable(archive-memory-test memory_test.c archive_memory.c)
target_compile_definitions(archive-memory-test PRIVATE ARCHIVE_MEMORY_BUDGET=1048576)
target_link_libraries(archive-memory-test PRIVATE liblzma)
if(MSVC)
  set_property(TARGET vesperwind-archive archive-space-test archive-memory-test archive_static zlibstatic liblzma PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
